ບົດທີ 1: ໄຟກະພິບດວງທຳອິດ (Digital Output & Blink)

(1) ຈຸດປະສົງຂອງບົດຮຽນ: ເຂົ້າໃຈໂຄງສ້າງຫຼັກຂອງໂຄ້ດ Arduino ຄື void setup() ແລະ void loop() ພ້ອມຮຽນຮູ້ການສັ່ງຈ່າຍໄຟ Digital Output (HIGH / LOW) ແລະ ການໜ່ວງເວລາດ້ວຍ delay().

(2) ອຸປະກອນທີ່ໃຊ້ໃນບົດນີ້: ບອດ Arduino UNO R3 1 ບອດ, ເບຣດບອດ 1 ແຜ່ນ, ຫຼອດ LED 1 ດວງ, ຕົວຕ້ານທານ 220Ω 1 ໂຕ, ສາຍຈຳເປີ 2 ເສັ້ນ.

(3) ໂຄ້ດຕົວຢ່າງພ້ອມຄຳອະທິບາຍພາສາລາວ:

int ledPin = 13; // ກຳນົດຕົວແປ ledPin ໃຫ້ເກັບໝາຍເລກຂາ 13

void setup() {
  // ຟັງຊັນ setup() ຈະເຮັດວຽກພຽງຄັ້ງດຽວ ຕອນເປີດໄຟ ຫຼື ກົດ Reset
  pinMode(ledPin, OUTPUT); // ຕັ້ງຄ່າຂາ 13 ເປັນ Output (ຂາສົ່ງໄຟອອກ)
}

void loop() {
  // ຟັງຊັນ loop() ຈະເຮັດວຽກວົນຊ້ຳໄປເລື້ອຍໆ ຈົນກວ່າຈະປິດໄຟ
  digitalWrite(ledPin, HIGH); // ສັ່ງຈ່າຍໄຟ 5V (HIGH) ໃຫ້ຫຼອດ LED ຕິດ
  delay(1000);                // ໜ່ວງເວລາ 1000 ມິນລິວິນາທີ (ເທົ່າກັບ 1 ວິນາທີ)

  digitalWrite(ledPin, LOW);  // ສັ່ງປິດໄຟ 0V (LOW) ໃຫ້ຫຼອດ LED ດັບ
  delay(1000);                // ໜ່ວງເວລາ 1 ວິນາທີ ກ່ອນວົນກັບໄປເປີດໃໝ່
}

int greenLed = 8;
int yellowLed = 9;
int redLed = 10;

void setup() {
  pinMode(greenLed, OUTPUT);  // ຕັ້ງຂາ 8 ເປັນ Output
  pinMode(yellowLed, OUTPUT); // ຕັ້ງຂາ 9 ເປັນ Output

}

void loop() {
  // ໄລຍະໄຟຂຽວ 3 ວິນາທີ

  digitalWrite(yellowLed, LOW);
  digitalWrite(redLed, LOW);
  delay(3000);

  // ໄລຍະໄຟເຫຼືອງ 1 ວິນາທີ
  digitalWrite(greenLed, LOW);

  digitalWrite(redLed, LOW);
  delay(1000);

  digitalWrite(greenLed, LOW);
  digitalWrite(yellowLed, LOW);
  digitalWrite(redLed, HIGH);

int trigPin = 5;
int echoPin = 6;
int buzzerPin = 7;

void setup() {
  Serial.begin(9600);         // ເປີດການສື່ສານຜ່ານ Serial Monitor ທີ່ຄວາມໄວ 9600
  pinMode(trigPin, OUTPUT);   // Trig ເປັນ Output
  pinMode(echoPin, INPUT);    // Echo ເປັນ Input
  pinMode(buzzerPin, OUTPUT); // Buzzer ເປັນ Output
}

void loop() {
  // ປ່ອຍສັນຍານກະຕຸ້ນທີ່ຂາ Trig
  digitalWrite(trigPin, LOW);
  delayMicroseconds(2);
  digitalWrite(trigPin, HIGH);
  delayMicroseconds(10);
  digitalWrite(trigPin, LOW);

  // ວັດໄລຍະເວລາທີ່ຂາ Echo ເປັນ HIGH (ໄປ-ກັບ)
  long duration = pulseIn(echoPin, HIGH);
  int distance = duration * 0.034 / 2; // ແປງເປັນໄລຍະທາງ (cm)

  Serial.print("Distance: ");
  Serial.print(distance);
  Serial.println(" cm");

  if (distance > 0 && distance < 15) { // ຖ້າມີວັດຖຸໃກ້ກວ່າ 15 cm
    digitalWrite(buzzerPin, HIGH);       // ສັ່ງ Buzzer ດັງ
  } else {
    digitalWrite(buzzerPin, LOW);        // ປິດສຽງ Buzzer
  }

  delay(200);
}

(3) ໂຄ້ດຕົວຢ່າງພ້ອມຄຳອະທິບາຍພາສາລາວ:

#include <Servo.h>

Servo gateServo;
int trigPin = 5;
int echoPin = 6;
int buzzerPin = 7;
int greenLed = 8;
int redLed = 10;

void setup() {
  gateServo.attach(3);
  pinMode(trigPin, OUTPUT);
  pinMode(echoPin, INPUT);
  pinMode(buzzerPin, OUTPUT);
  pinMode(greenLed, OUTPUT);
  pinMode(redLed, OUTPUT);

  // ສະຖານະເລີ່ມຕົ້ນ: ປິດໄມ້ກັ້ນ/ຝາ ແລະ ເປີດໄຟແດງ
  gateServo.write(0);
  digitalWrite(redLed, HIGH);
  digitalWrite(greenLed, LOW);
}

void loop() {
  digitalWrite(trigPin, LOW);
  delayMicroseconds(2);
  digitalWrite(trigPin, HIGH);
  delayMicroseconds(10);
  digitalWrite(trigPin, LOW);

  long duration = pulseIn(echoPin, HIGH);
  int distance = duration * 0.034 / 2;

  if (distance > 0 && distance < 15) {
    // ປ່ຽນເປັນສະຖານະເປີດ
    digitalWrite(redLed, LOW);
    digitalWrite(greenLed, HIGH);

    // ສົ່ງສຽງແຈ້ງເຕືອນສັ້ນໆ 150 ms
    digitalWrite(buzzerPin, HIGH);
    delay(150);
    digitalWrite(buzzerPin, LOW);

    gateServo.write(90); // ໝຸນເປີດ 90 ອົງສາ
    delay(3000);         // ຄ້າງໄວ້ 3 ວິນາທີ

    // ກັບຄືນສູ່ສະຖານະປິດ
    gateServo.write(0);
    digitalWrite(greenLed, LOW);
    digitalWrite(redLed, HIGH);
  }

  delay(100);
}
